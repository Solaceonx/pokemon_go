env OMP_NUM_THREADS=1 OPENBLAS_NUM_THREADS=1 VECLIB_MAXIMUM_THREADS=1 MKL_NUM_THREADS=1 /Users/marco/Developer/pokemon_go/.venv/bin/python - <<'PY'
import importlib.util
import random
from pogo.agents import Planner
from pogo.battle import Battle
from pogo.gamedata import GameData
from pogo.learner import Learner, QNet
spec = importlib.util.spec_from_file_location('pogo.task20_before', '/tmp/task20-learner-before.py')
before = importlib.util.module_from_spec(spec)
spec.loader.exec_module(before)
q = QNet.load('runs/general-decide/q.npz')
pool = GameData().meta(50)
rng = random.Random(20040)
for seed in range(100):
    teams = rng.sample(pool, 3), rng.sample(pool, 3)
    a = Battle(*teams, before.Learner(q=q, seed=seed), Planner(seed=seed + 1), seed=seed, record=True)
    b = Battle(*teams, Learner(q=q, seed=seed), Planner(seed=seed + 1), seed=seed, record=True)
    assert a.run() == b.run() and a.log == b.log, seed
print('100 of 100 old-model battle logs identical to pre-edit Learner')
PY
