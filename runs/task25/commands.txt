import os
_original_nice = os.nice
def _retain_low_priority(increment):
    try:
        return _original_nice(increment)
    except PermissionError:
        current = os.getpriority(os.PRIO_PROCESS, 0)
        if increment >= 0 and current >= 10:
            return current
        raise
os.nice = _retain_low_priority
